\begin{circuitikz}
  % four toggle flip-flops: each one's inverted output clocks the next
  \foreach \i in {0,...,3} {
    \draw (3.2*\i,0) node[flipflop T] (F\i) {};
    \draw (F\i.pin 1) -- ++(-0.3,0) coordinate (t\i) -- ++(0,0.9) coordinate (r\i);
    \draw (F\i.pin 6) -- ++(0.3,0) to[short, -o] ++(0,1.6) node[above] {$Q_{\i}$};
  }
  \draw (r3) -- (r0) to[short, -o] ++(-0.6,0) node[left] {$1$};
  \foreach \i in {0,1,2} \draw (r\i) node[circ] {};
  \draw (F0.pin 3) to[short, -o] ++(-0.9,0) node[left] {$\mathit{CLK}$};
  \foreach \i [evaluate=\i as \j using int(\i+1)] in {0,1,2} \draw (F\i.pin 4) -- (F\j.pin 3);
\end{circuitikz}
