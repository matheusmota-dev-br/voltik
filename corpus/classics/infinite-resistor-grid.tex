\begin{circuitikz}
  \ctikzset{bipoles/length=0.9cm}
  % a 5 × 4 window of the infinite square lattice: every edge a resistor R
  \foreach \y in {0,...,3} \foreach \x in {0,...,3}
    \draw (2*\x,2*\y) to[R] (2*\x+2,2*\y);
  \foreach \x in {0,...,4} \foreach \y in {0,...,2}
    \draw (2*\x,2*\y) to[R] (2*\x,2*\y+2);
  % the lattice goes on in every direction
  \foreach \y in {0,...,3} {
    \draw[dashed] (0,2*\y) -- ++(-0.8,0);
    \draw[dashed] (8,2*\y) -- ++(0.8,0);
  }
  \foreach \x in {0,...,4} {
    \draw[dashed] (2*\x,0) -- ++(0,-0.8);
    \draw[dashed] (2*\x,6) -- ++(0,0.8);
  }
  \foreach \y in {0,...,3} \foreach \x in {0,...,4}
    \node[circ] at (2*\x,2*\y) {};
  % two neighbours: current I in at A spreads as I/4 on each of A's four edges, and the same
  % current out at B gathers I/4 on each of B's: the edge AB carries I/2
  \node[circ, red, scale=1.6] at (4,2) {};
  \node[circ, red, scale=1.6] at (4,4) {};
  \node[red, below right] at (4,2) {$A$};
  \node[red, above right] at (4,4) {$B$};
  \draw[red, thick] (4,2) -- (4,2.55) (4,3.45) -- (4,4);
  \node[right, align=left] at (9.4,3) {every edge: $R$\\[4pt] $\displaystyle R_{AB} = \frac{R}{2}$};
\end{circuitikz}
